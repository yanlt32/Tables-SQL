1. EXTRACT (Extração dos dados OLTP)

-- Extrair alunos
SELECT a.pk_aluno, p.nome_pessoa, a.ra, p.cpf, p.data_nascimento, a.data_ingresso
FROM tb_alunos a JOIN tb_pessoas p ON a.fk_pessoa = p.pk_pessoa;

-- Extrair cursos
SELECT pk_curso, nome_curso, duracao_semestres FROM tb_cursos;

-- Extrair datas
SELECT data_vencimento FROM tb_mensalidades UNION SELECT data_pagamento FROM tb_pagamentos;

-- Extrair dados financeiros
SELECT m.pk_mensalidade, m.valor, m.data_vencimento, pg.valor_pago, pg.data_pagamento
FROM tb_mensalidades m LEFT JOIN tb_pagamentos pg ON pg.fk_mensalidade = m.pk_mensalidade;

2. TRANSFORM (Transformação dos dados)

-- Transformação 1: Calcular faixa etária
CASE
  WHEN TIMESTAMPDIFF(YEAR, data_nascimento, CURDATE()) < 20 THEN 'Menos de 20 anos'
  WHEN TIMESTAMPDIFF(YEAR, data_nascimento, CURDATE()) < 30 THEN '20 a 29 anos'
  ELSE '30 anos ou mais'
END AS faixa_etaria;

-- Transformação 2: Extrair componentes de data
YEAR(d) AS ano, MONTH(d) AS mes,
IF(MONTH(d) <= 6, 1, 2) AS semestre,
ELT(MONTH(d), 'Janeiro','Fevereiro',...) AS nome_mes;

-- Transformação 3: Tratar valores nulos
COALESCE(pg.valor_pago, 0.00) AS valor_pago
